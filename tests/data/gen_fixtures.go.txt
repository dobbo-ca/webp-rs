package main

import (
	"image"
	"image/color"
	"image/png"
	"math"
	"math/rand"
	"os"
)

func save(name string, m image.Image) {
	f, _ := os.Create(name)
	e := png.Encoder{CompressionLevel: png.BestCompression}
	e.Encode(f, m)
	f.Close()
}

type blob struct{ x, y, r, a float64 }

func blobs(rng *rand.Rand, n int, rmin, rmax float64) []blob {
	b := make([]blob, n)
	for i := range b {
		b[i] = blob{rng.Float64() * 768, rng.Float64() * 768, rmin + rng.Float64()*(rmax-rmin), 0.5 + rng.Float64()*0.5}
	}
	return b
}

func main() {
	rng := rand.New(rand.NewSource(1))
	// Brightfield H&E look: white background, purple nuclei, pink cytoplasm.
	he := image.NewRGBA(image.Rect(0, 0, 768, 768))
	cyto := blobs(rng, 40, 60, 160)
	nuc := blobs(rng, 300, 5, 14)
	for y := 0; y < 768; y++ {
		for x := 0; x < 768; x++ {
			r, g, b := 245.0, 240.0, 245.0
			for _, c := range cyto {
				d := math.Hypot(float64(x)-c.x, float64(y)-c.y) / c.r
				if d < 1 {
					w := c.a * (1 - d*d)
					r -= w * 20; g -= w * 90; b -= w * 40
				}
			}
			for _, c := range nuc {
				d := math.Hypot(float64(x)-c.x, float64(y)-c.y) / c.r
				if d < 1 {
					w := c.a * (1 - d*d*d)
					r -= w * 150; g -= w * 190; b -= w * 100
				}
			}
			n := 0.0
			he.Set(x, y, color.RGBA{clamp(r+n)&^7, clamp(g+n)&^7, clamp(b+n)&^7, 255})
		}
	}
	save("he_768.png", he)
	// Fluorescence: dark background, bright nuclei with soft edges.
	fl := image.NewGray(image.Rect(0, 0, 768, 768))
	fn := blobs(rng, 120, 6, 22)
	for y := 0; y < 768; y++ {
		for x := 0; x < 768; x++ {
			v := 12.0
			for _, c := range fn {
				d := math.Hypot(float64(x)-c.x, float64(y)-c.y) / c.r
				if d < 1.3 {
					v += c.a * 230 * math.Exp(-2.5*d*d)
				}
			}
			
			fl.SetGray(x, y, color.Gray{clamp(v)})
		}
	}
	save("fl_768.png", fl)
}

func clamp(v float64) uint8 {
	if v < 0 { return 0 }
	if v > 255 { return 255 }
	return uint8(v + 0.5)
}
